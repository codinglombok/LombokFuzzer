cmake_minimum_required(VERSION 3.14)
project(lombokfuzzer VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(lombokfuzzer src/lombokfuzzer.cpp)
target_include_directories(lombokfuzzer PUBLIC include/)
enable_testing()
add_executable(lombokfuzzer_test tests/test_main.cpp)
target_link_libraries(lombokfuzzer_test lombokfuzzer)
add_test(NAME lombokfuzzer_test COMMAND lombokfuzzer_test)
